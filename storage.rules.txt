rules_version = '2';

// =====================================================================
// Bazar Dzair — Firebase Storage Security Rules (نسخة معززة)
// انسخ هذا الملف كاملاً إلى Firebase Console > Storage > Rules ثم انشر.
//
// ⚠️ قبل النشر: بدّلي القيمة 'ADMIN_EMAIL_HERE' بإيميل تسجيل الدخول
//    الحقيقي تاعك (نفس القيمة المستعملة في firestore.rules).
// =====================================================================

service firebase.storage {
  match /b/{bucket}/o {

    match /products/{productId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null
                   && request.auth.token.email == 'ADMIN_EMAIL_HERE'
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/.*');
      allow delete: if request.auth != null
                    && request.auth.token.email == 'ADMIN_EMAIL_HERE';
    }

    // أي مسار آخر مرفوض افتراضيًا.
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
