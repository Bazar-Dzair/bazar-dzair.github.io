// حاليًا عبر Cloudinary مباشرة من admin.html (uploadToCloudinary) — بلا
// أي علاقة بهاذ الملف أو بـ Firebase Storage.
// هاذ الملف محفوظ فقط كمرجع جاهز: إذا حبيت مستقبلاً تفعّل خطة Blaze وتحول

// Bazar Dzair — Firebase Storage Security Rules (نسخة معززة v2)
// انسخ هذا الملف كاملاً إلى Firebase Console > Storage > Rules ثم انشر
// (أو فعّل نشره التلقائي — راجع firebase.json و تعليمات_نشر التي بالأسفل).
//
// ⚠️ كانت القيمة هنا لا تزال 'ADMIN_UID_HERE' (لم تُستبدل أبدًا) بينما
//    firestore.rules يستعمل UID حقيقيًا — ما كان يمنع رفع شعار/بانر المتجر من
//    admin.html نهائيًا (isAdmin() كانت ترجع false دائمًا). تم الآن مزامنة
//    القيمة مع نفس UID المستخدم في firestore.rules. إذا تغيّر حساب الأدمن مستقبلاً
//    يجب تحديث القيمة هنا وفي firestore.rules معًا (Firebase Console → Authentication
//    → Users → User UID).
// =====================================================================

service firebase.storage {
  match /b/{bucket}/o {

    function isAdmin() {
      return request.auth != null
             && request.auth.uid == 'GOBngnCP2eMTLZrJpf72GOmXvvO2';
    }

    // صور المنتجات — قراءة عامة، الرفع/الحذف/التعديل للأدمن فقط.
    match /products/{productId}/{fileName} {
      allow read: if true;
      allow write: if isAdmin()
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/(jpeg|png|webp)');
      allow delete: if isAdmin();
    }

    // شعار وبانر المتجر (يُرفعان من admin.html) — نفس القيود.
    match /site/{fileName} {
      allow read: if true;
      allow write: if isAdmin()
                   && request.resource.size < 5 * 1024 * 1024
                   && request.resource.contentType.matches('image/(jpeg|png|webp)');
      allow delete: if isAdmin();
    }

    // أي مسار آخر (بما فيه /admin/, /system/, /private/ أو أي مسار غير معروف) مرفوض افتراضيًا.
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
