rules_version = '2';

// =====================================================================
// Bazar Dzair — Firestore Security Rules (نسخة معززة v2)
// انسخ هذا الملف كاملاً إلى Firebase Console > Firestore > Rules ثم انشر.
//
// ⚠️ قبل النشر: بدّل القيمة 'GOBngnCP2eMTLZrJpf72GOmXvvO2' بـ UID حساب الأدمن الحقيقي.
//    كيفية الحصول عليه: Firebase Console → Authentication → Users →
//    عمود "User UID" لحسابك (وليس الإيميل).
//    استخدام UID أكثر أمانًا من الإيميل لأنه غير قابل للتخمين ولا يكشف
//    عنوان تسجيل الدخول الحقيقي في حال تسريب الكود.
// =====================================================================

service cloud.firestore {
  match /databases/{database}/documents {

    // فقط صاحب هذا الـ UID يُعتبر أدمن — وليس أي حساب Firebase مسجل دخول.
    function isAdmin() {
      return request.auth != null
             && request.auth.uid == 'GOBngnCP2eMTLZrJpf72GOmXvvO2';
    }

    // المنتجات — تُعرض لأي زائر، الكتابة للأدمن فقط.
    match /products/{id} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    // التصنيفات — تُعرض لأي زائر، الكتابة للأدمن فقط.
    match /categories/{id} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    // التقييمات — كل تقييم حقيقي فقط. أي زائر (بدون تسجيل دخول) يقدر يضيف
    // تقييمًا جديدًا لمنتج موجود فعلاً، لكن يُحفظ دائمًا بحالة "قيد المراجعة"
    // (approved == false إجباريًا عند الإنشاء) ولا يظهر للزبائن إلا بعد موافقة
    // الأدمن من لوحة التحكم. القراءة العامة تقتصر على التقييمات المنشورة فقط؛
    // الأدمن وحده يقرأ/ينشر/يحذف أي تقييم (بما فيها قيد المراجعة).
    function isValidReview(d) {
      return
        d.keys().hasOnly(['productId','productName','authorName','ratingValue','text','approved','createdAt']) &&
        d.keys().hasAll(['productId','authorName','ratingValue','text','approved','createdAt']) &&
        // التاريخ يضعه الخادم فقط (serverTimestamp)
        d.createdAt == request.time &&
        d.approved == false &&
        d.productId is string && d.productId.size() > 0 && d.productId.size() < 200 &&
        exists(/databases/$(database)/documents/products/$(d.productId)) &&
        d.authorName is string && d.authorName.size() >= 2 && d.authorName.size() <= 60 &&
        d.text is string && d.text.size() >= 3 && d.text.size() <= 600 &&
        d.ratingValue is int && d.ratingValue >= 1 && d.ratingValue <= 5 &&
        (!('productName' in d) || (d.productName is string && d.productName.size() < 200));
    }
    match /reviews/{id} {
      allow read: if resource.data.approved == true || isAdmin();
      allow create: if isValidReview(request.resource.data);
      allow update, delete: if isAdmin();
    }

    // الإعدادات — كل مستند settings/{id} قراءته عامة (اسم المتجر، الشعار،
    // أسعار التوصيل...)، ما عدا settings/email الذي يحتوي بيانات SMTP
    // الحساسة (smtpHost/smtpUser/smtpPassword) والذي يبقى Admin-only حتى
    // في القراءة. الكتابة دائمًا Admin فقط.
    match /settings/{id} {
      allow read: if id != 'email' || isAdmin();
      allow write: if isAdmin();
    }

    // ✅ الطلبات — مساران لإنشاء الطلب:
    //   1) الأساسي: عبر Cloudflare Worker (مسار /create-order) بصلاحيات Service Account
    //      (تتجاوز هذه القواعد). هو الوحيد الذي يرى عنوان IP الحقيقي، فيطبّق حظر الـ IP.
    //   2) الاحتياطي: كتابة مباشرة من المتصفح، مسموحة فقط ما دام "الوضع الصارم" مُطفأً
    //      وبشرط اجتياز isValidOrder() (شكل البيانات، الهاتف غير محظور، السعر حقيقي...).
    //      هذا المسار يضمن ألا تتوقف الطلبات أبدًا لو تعطّل الـ Worker أو نُشر بشكل ناقص.
    // "الوضع الصارم" = وثيقة settings/security بحقل strictOrders == true، يفعّله الأدمن من
    // لوحة التحكم (تبويب المحظورون) بعد التأكد أن الـ Worker سليم. حينها يُمنع الإنشاء
    // المباشر تمامًا فيصبح حظر الـ IP مُلزِمًا (لا يمكن تجاوزه بالكتابة المباشرة في Firestore).
    // الأدمن وحده يقرأ ويعدّل ويحذف الطلبات.
    function strictOrders() {
      return exists(/databases/$(database)/documents/settings/security)
             && get(/databases/$(database)/documents/settings/security).data.get('strictOrders', false) == true;
    }
    match /orders/{id} {
      allow create: if isAdmin() || (!strictOrders() && isValidOrder(request.resource.data));
      allow read, update, delete: if isAdmin();
    }

    // ⚠️ أرقام الهواتف المحظورة — كانت هذه القاعدة ناقصة تمامًا (لا يوجد أي match
    // لهذا الـ collection)، فكانت قاعدة الرفض الافتراضي في نهاية الملف تمنع لوحة
    // التحكم (admin.html: bazarLoadBlockedPhones/blockPhone/unblockPhone) من قراءة
    // أو كتابة أي رقم محظور رغم تسجيل دخول الأدمن — يعمل الـ Worker فقط لأنه يستخدم
    // Service Account الذي يتجاوز Security Rules أصلاً.
    match /blockedPhones/{phone} {
      allow read, write: if isAdmin();
    }

    // 🚫 عناوين IP المحظورة — معرّف الوثيقة = مفتاح الـ IP (IPv4 كما هو، أو أول 64 بت
    // من IPv6 على شكل v6-2001-db8-1-2). الأدمن وحده يقرأ ويكتب من لوحة التحكم
    // (admin.html)، والـ Worker يقرأها بصلاحيات Service Account قبل قبول أي طلب. لا
    // قراءة عامة إطلاقًا حتى لا يعرف المزعج أن IP الخاص به محظور من هذه القائمة.
    match /blockedIps/{key} {
      allow read, write: if isAdmin();
    }

    // isValidOrder(): شروط الطلب في المسار الاحتياطي (الكتابة المباشرة). الـ Worker يعيد نفس
    // الشروط يدويًا (create-order.js › validateOrder) قبل الكتابة — أي تعديل هنا يُطبَّق هناك أيضًا.
    function isValidOrder(d) {
      return
        // الحقول المسموحة فقط — أي حقل إضافي (adminNotes، role، isAdmin...) يرفض الطلب
        d.keys().hasOnly(['customerName','customerPhone','wilaya','address','product','productId','quantity','price','shipping','deliveryType','shippingCompany','total','status','createdAt']) &&
        // ✅ productId أصبح إجباريًا (وليس اختياريًا) لسد ثغرة كانت تسمح بتجاوز
        // التحقق من السعر الحقيقي عبر إرسال طلب بدون productId.
        d.keys().hasAll(['customerName','customerPhone','wilaya','address','product','productId','quantity','price','total','status']) &&

        // status يجب أن يكون قيمة البداية الثابتة فقط — لا يقدر الزبون يبعث طلب "مؤكد" مباشرة
        d.status == 'جديد' &&

        // نصوص أساسية بحدود طول منطقية (تمنع payloads ضخمة أو حقول فارغة)
        d.customerName is string && d.customerName.size() > 0 && d.customerName.size() < 100 &&
        d.customerPhone is string && d.customerPhone.matches('^0[5-7][0-9]{8}$') &&
        // 🚫 رقم محظور من طرف الأدمن (collection blockedPhones) لا يقدر يسجّل طلب جديد.
        !exists(/databases/$(database)/documents/blockedPhones/$(d.customerPhone)) &&
        d.wilaya is string && d.wilaya.size() > 0 && d.wilaya.size() < 100 &&
        d.address is string && d.address.size() > 0 && d.address.size() < 300 &&
        d.product is string && d.product.size() > 0 && d.product.size() < 200 &&
        d.productId is string && d.productId.size() > 0 && d.productId.size() < 200 &&
        (!('deliveryType' in d) || (d.deliveryType is string && d.deliveryType.size() < 50)) &&
        (!('shippingCompany' in d) || (d.shippingCompany is string && d.shippingCompany.size() < 100)) &&

        // أرقام منطقية — يمنع كمية سالبة أو صفر أو كمية ضخمة (بوت)، وسعر/مجموع سالب
        d.quantity is int && d.quantity > 0 && d.quantity <= 50 &&
        d.price is number && d.price >= 0 &&
        d.total is number && d.total >= 0 &&
        (!('shipping' in d) || (d.shipping is number && d.shipping >= 0)) &&

        // ✅ حماية من التلاعب بالمجموع: total يجب أن يساوي حسابيًا price×quantity + shipping
        d.total == (d.price * d.quantity) + (('shipping' in d) ? d.shipping : 0) &&

        // ✅ حماية من التلاعب بالسعر: productId إجباري الآن، ويجب أن يطابق price
        // السعر الحقيقي المخزّن فعليًا في المنتج — هذا يمنع مهاجم يتواصل مباشرة مع
        // Firestore (بدون المرور بواجهة الموقع) من إرسال سعر مزيف أو منتج وهمي غير موجود.
        get(/databases/$(database)/documents/products/$(d.productId)).data.price == d.price;
    }

    // Deny-by-default صريح: أي collection أو مستند غير معروف/غير مذكور أعلاه ممنوع تمامًا.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
